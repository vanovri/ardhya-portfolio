    // Patch #1: Dynamic Sitemap rute aktual /magang-nasional/* + filter fail-closed ACTIVE + sharding > 50K
import type { MetadataRoute } from "next";
import { evaluateJobLifecycle, type JobRecord } from "../lib/job-lifecycle";

const BASE_URL = "https://maganghub.kemnaker.go.id";
const INTERNAL_API_URL = process.env.INTERNAL_API_URL; // Hindari self-HTTP fetch saat build
const SHARD_LIMIT = 50_000;

export const STATIC_ROUTES = [
  "/",
  "/magang-nasional/lowongan",
  "/magang-nasional/penyelenggara",
  "/magang-nasional/persyaratan",
  "/magang-nasional/tentang",
  "/magang-nasional/faq",
  "/pendaftaran",
  "/pendaftaran/magang-nasional",
] as const;

export async function generateSitemaps(): Promise<Array<{ id: number }>> {
  const count = await fetchActiveJobCount();
  const shards = Math.max(1, Math.ceil((count + STATIC_ROUTES.length) / SHARD_LIMIT));
  return Array.from({ length: shards }, (_, id) => ({ id }));
}

export default async function sitemap({ id = 0 } = {}): Promise<MetadataRoute.Sitemap> {
  const now = new Date();
  const staticUrls: MetadataRoute.Sitemap = id === 0
    ? STATIC_ROUTES.map((p) => ({
        url: new URL(p, BASE_URL).toString(),
        lastModified: now,
        changeFrequency: p === "/magang-nasional/lowongan" ? "hourly" : "daily",
        priority: p === "/" ? 1.0 : 0.85,
      }))
    : [];

  if (!INTERNAL_API_URL) return staticUrls;
  const jobs = await fetchActiveJobs(id, SHARD_LIMIT);
  const jobUrls: MetadataRoute.Sitemap = jobs
    .filter((job) => evaluateJobLifecycle(job, now) === "ACTIVE")
    .map((job) => ({
      url: `${BASE_URL}/magang-nasional/lowongan/${encodeURIComponent(job.slug)}`,
      lastModified: new Date(job.updatedAt),
      changeFrequency: "daily",
      priority: 0.8,
    }));

  return [...staticUrls, ...jobUrls].slice(0, SHARD_LIMIT);
}

    // Patch #3: JobPosting JSON-LD (ONSITE/HYBRID/REMOTE, multi-lokasi, fail-closed, XSS-safe)
import { evaluateJobLifecycle, type JobRecord } from "../../lib/job-lifecycle";
import { buildJobPostingSchema, serializeJsonLdSafe } from "../../lib/job-schema";

export function JobPostingJsonLd({
  job,
  now = new Date(),
}: {
  job: JobRecord;
  now?: Date;
}) {
  // Defense-in-depth: kembalikan null jika lifecycle !== ACTIVE
  if (evaluateJobLifecycle(job, now) !== "ACTIVE") {
    return null;
  }

  const schema = buildJobPostingSchema(job, now);
  if (!schema) return null;

  return (
    <script
      type="application/ld+json"
      dangerouslySetInnerHTML={{
        // Escapes <, >, &, \u2028, \u2029 mencegah script-breakout XSS
        __html: serializeJsonLdSafe(schema),
      }}
    />
  );
}

    // Patch #4: Hybrid Expired Lifecycle (200 noindex -> 410 Gone + X-Robots-Tag)
// Catatan Next.js 16+: migrasikan ekspor middleware -> proxy.ts bila upgrade ke Next 16+.
import { NextResponse, type NextRequest } from "next/server";
import { extractJobDetailSlug, evaluateJobLifecycle } from "./lib/job-lifecycle";
import { renderGoneHtml } from "./lib/gone-page";

export async function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;

  // Tambahkan X-Robots-Tag: noindex pada legacy endpoint 410
  if (pathname === "/be/v1/api/list/vacancies-active") {
    return new NextResponse(JSON.stringify({ status: 410, error: "Gone" }), {
      status: 410,
      headers: {
        "Content-Type": "application/json",
        "X-Robots-Tag": "noindex, follow",
        "Cache-Control": "public, max-age=3600, s-maxage=86400",
      },
    });
  }

  // Anti-tabrakan: abaikan /kota/*, /bidang/*, /perusahaan/*
  const slug = extractJobDetailSlug(pathname);
  if (!slug) return NextResponse.next();

  const state = await lookupArchiveStateWithTimeout(slug, 400);
  if (state === "ARCHIVED_GONE") {
    return new NextResponse(renderGoneHtml(slug), {
      status: 410,
      headers: {
        "Content-Type": "text/html; charset=utf-8",
        "X-Robots-Tag": "noindex, follow",
        "Cache-Control": "public, max-age=3600, s-maxage=86400",
      },
    });
  }

  return NextResponse.next();
}
